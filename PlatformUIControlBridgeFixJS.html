(function () { "use strict"; const platform = window.PersonnelPlatform; if (!platform) return; function text(value) { return String(value == null ? "" : value).trim(); } function refreshPersonnelUI() { if (typeof window.renderPreviewTable === "function") { window.renderPreviewTable(); } if (typeof window.updateAoCartCount_ === "function") { window.updateAoCartCount_(); } if (typeof window.updateReportGeneratorButton_ === "function") { window.updateReportGeneratorButton_(); } } function recordForPreviewControl(control) { const row = control.closest("#previewTableBody tr"); const body = row?.parentElement; if (!row || !body) return null; const rows = [...body.querySelectorAll(":scope > tr")]; const index = rows.indexOf(row); if (index < 0) return null; const records = Array.isArray(window.PersonnelState?.filteredRecords) ? window.PersonnelState.filteredRecords : []; return records[index] || null; } function isAoSelectionControl(node) { if (!node) return false; if (node.matches?.("button, [role='button']")) return true; return node.classList?.contains("ao-selection-cell") || false; } // The current table renderer places the AO action on the table cell itself, // not always on a nested button.
