//---------------------------------- // Reassignment Workspace Remove Controls //---------------------------------- document.addEventListener( "DOMContentLoaded", initializeReassignmentWorkspaceClear_ ); function initializeReassignmentWorkspaceClear_() { const attach = () => { const actions = document.querySelector( ".reassignment-source-panel .source-actions" ); if (!actions) { return false; } let button = document.getElementById( "clearAllSelectedPersonnelButton" ); if (!button) { button = document.createElement("button"); button.id = "clearAllSelectedPersonnelButton"; button.type = "button"; button.className = "button button-secondary reassignment-remove-all-button"; button.addEventListener( "click", clearAllSelectedPersonnel_ ); const clearFiltersButton = [ ...actions.querySelectorAll("button"), ].find(item => item.textContent.trim().toLowerCase() === "clear filters" ); if (clearFiltersButton) { actions.insertBefore(button, clearFiltersButton); } else { actions.appendChild(button); } } button.textContent = "Remove All"; button.title = "Remove all personnel currently under To Be Assigned"; return true; }; if (!attach()) { const observer = new MutationObserver(() => { if (attach()) { observer.disconnect(); } }); observer.observe(document.body, { childList: true, subtree: true, }); } patchReassignmentSourcePersonnelRemoval_(); injectReassignmentRemovalStyles_(); } function clearAllSelectedPersonnel_() { const unassignedKeys = [ ...PersonnelState.aoCart.keys(), ].filter(key => !ReassignmentState.assignments.has(key) ); if (!unassignedKeys.length) { return; } if (!confirm( `Remove all ${unassignedKeys.length} personnel under To Be Assigned?\n\nAssigned personnel in Step 3 will remain.` )) { return; } unassignedKeys.forEach(key => { PersonnelState.aoCart.delete(key); ReassignmentState.selectedKeys.delete(key); }); persistReassignmentRemovalChanges_(); } function removePersonnelFromReassignment_(key) { if (!PersonnelState.aoCart.has(key)) { return; } // The individual X appears only in Step 1, so this removes only an // unassigned personnel record. Assigned personnel remain controlled by // their existing Step 3 X button, which returns them to Step 1. PersonnelState.aoCart.delete(key); ReassignmentState.selectedKeys.delete(key); persistReassignmentRemovalChanges_(); } function persistReassignmentRemovalChanges_() { if (typeof saveAoCart_ === "function") { saveAoCart_(); } if (typeof saveReassignmentState_ === "function") { saveReassignmentState_(); } if (typeof updateAoCartCount_ === "function") { updateAoCartCount_(); } if (typeof updateReportGeneratorButton_ === "function") { updateReportGeneratorButton_(); } if (typeof renderPreviewTable === "function") { renderPreviewTable(); } if (typeof renderReassignmentWorkspace_ === "function") { renderReassignmentWorkspace_(); } } function patchReassignmentSourcePersonnelRemoval_() { renderSourcePersonnel_ = function(entries) { const list = document.getElementById( "reassignmentSourceList" ); list.replaceChildren(); if (!entries.length) { list.appendChild( makeText_("div", "No personnel match the filters.") ); return; } entries.forEach(([key, person]) => { const row = makeElement_( "div", "reassignment-person-row" ); const checkbox = document.createElement("input"); checkbox.type = "checkbox"; checkbox.checked = ReassignmentState.selectedKeys.has(key); checkbox.addEventListener("change", () => { if (checkbox.checked) { ReassignmentState.selectedKeys.add(key); } else { ReassignmentState.selectedKeys.delete(key); } updateReassignmentControls_( PersonnelState.aoCart.size ); }); const details = document.createElement("div"); details.className = "reassignment-person-details"; details.append( makeText_( "strong", [person.rank, person.fullName] .filter(Boolean) .join(" ") ), makeText_( "small", person.office || "No current office" ) ); const remove = document.createElement("button"); remove.type = "button"; remove.className = "reassignment-source-remove-button"; remove.textContent = "×"; remove.title = "Remove this personnel from To Be Assigned"; remove.setAttribute( "aria-label", `Remove ${person.fullName || "personnel"}` ); remove.addEventListener("click", event => { event.preventDefault(); event.stopPropagation(); removePersonnelFromReassignment_(key); }); row.append(checkbox, details, remove); list.appendChild(row); }); }; } function injectReassignmentRemovalStyles_() { if (document.getElementById("reassignmentRemovalStyles")) { return; } const style = document.createElement("style"); style.id = "reassignmentRemovalStyles"; style.textContent = ` .reassignment-person-row { display: grid; grid-template-columns: auto minmax(0, 1fr) auto; align-items: center; gap: 10px; } .reassignment-person-details { min-width: 0; display: grid; } .reassignment-source-remove-button { width: 30px; height: 30px; display: inline-grid; place-items: center; border: 1px solid rgba(153, 39, 39, .2); border-radius: 999px; background: #fff5f5; color: #a32626; font: inherit; font-size: 20px; line-height: 1; cursor: pointer; } .reassignment-source-remove-button:hover { border-color: #a32626; background: #ffe4e4; } .reassignment-remove-all-button { color: #8b2020; } /* Keep Step 3 compact even when hundreds of personnel are assigned. */ .reassignment-results-panel { min-height: 0; align-self: stretch; } .reassignment-results-panel .assignment-card-grid { max-height: calc(100vh - 250px); min-height: 180px; overflow-y: auto; overscroll-behavior: contain; padding-right: 6px; scrollbar-gutter: stable; } .reassignment-results-panel .assignment-card-grid::-webkit-scrollbar { width: 10px; } .reassignment-results-panel .assignment-card-grid::-webkit-scrollbar-thumb { border: 2px solid transparent;
