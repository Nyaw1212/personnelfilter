response : {}; const records = Array.isArray(source.records) ? source.records : []; const nameEngine = window.PersonnelPlatform?.engines?.get?.("nameHandler") || window.$PF?.name; if (!nameEngine) { console.warn( "PersonnelAPI returned raw names because NameHandlerEngine was unavailable." ); return { ...source, records }; } const normalizedRecords = records.map(record => { const rawFullName = String(record?.["Full Name"] || "").trim(); const normalizedParts = nameEngine.normalize(record); const canonicalName = nameEngine.canonicalFullName(record, false) || rawFullName; const canonicalNameWithRank = nameEngine.canonicalFullName(record, true) || [record?.Rank, canonicalName].filter(Boolean).join(" "); return { ...record, // Keep the original table/search contract unchanged.
