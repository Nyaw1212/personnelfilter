//---------------------------------- // Reassignment Source Office Patch //---------------------------------- function getReassignmentPersonValue_(person, keys) { const sources = [ person || {}, person && person.record ? person.record : {}, ]; for (const source of sources) { for (const key of keys) { const value = source[key]; if ( value !== null && value !== undefined && String(value).trim() !== "" ) { return String(value).trim(); } } } return ""; } function buildReassignmentGeneratePayload_() { const record = findSelectedOfficeRecord_(); const role = ReassignmentGenerateState.selectedRole; if (!record) { throw new Error( "Select a valid signing office from the directory." ); } const signingOfficer = role === "DEPUTY" ? record.deputyName : record.chiefName; const signingPosition = role === "DEPUTY" ? record.deputyPosition : record.chiefPosition; if (!signingOfficer || !signingPosition) { throw new Error( `The selected office does not have a complete ${role.toLowerCase()} record.` ); } const missingSourceOffices = []; const personnel = [ ...ReassignmentState.assignments.entries(), ].map(([key, destinationOffice], index) => { const person = PersonnelState.aoCart.get(key); if (!person) { throw new Error( "A selected personnel record is no longer available." ); } const rank = getReassignmentPersonValue_(person, [ "rank", "Rank", "RANK", ]); const fullName = getReassignmentPersonValue_(person, [ "fullName", "Full Name", "FULL NAME", ]); const sourceOffice = getReassignmentPersonValue_(person, [ "office", "Office", "OFFICE", "currentOffice", "Current Office", ]); if (!sourceOffice) { missingSourceOffices.push( `${index + 1}. ${[rank, fullName].filter(Boolean).join(" ") || "Unknown personnel"}` ); } return { rank, fullName, // Do not block report generation when the Personnel sheet has no Office. // N/A is used so both the bridge and the document generator can continue. from: sourceOffice || "N/A", to: String(destinationOffice || "").trim(), notes: "", }; }); if (missingSourceOffices.length) { console.warn( "Generating reassignment report with missing source offices:", missingSourceOffices ); } return { orderNumber: document .getElementById("reassignmentOrderNumber") .value.trim(), orderDate: document .getElementById("reassignmentOrderDate") .value, signingOfficer, signingPosition, fileReference: document .getElementById("reassignmentFileReference") .value.trim(), personnel, }; }
