// Reassignment V1 — state-driven group destination change // Owns the Change action before Phase 3 and forces a clean plugin rerender. (function () { "use strict"; const platform = window.PersonnelPlatform; const engine = platform?.engines?.get?.("assignmentCore"); const plugin = window.$PF?.reassignmentV1; if (!platform || !engine || !plugin) return; const SCOPE = "reassignment-v1"; function text(value) { return String(value == null ? "" : value).trim(); } function rerender() { // open() synchronizes participants and renders entirely from engine state. plugin.close(); plugin.open(); } document.addEventListener("click", event => { const button = event.target.closest?.("[data-rp1-group-change]"); if (!button) return; event.preventDefault(); event.stopPropagation(); event.stopImmediatePropagation(); const groupId = text(button.dataset.rp1GroupChange); const group = button.closest(".rp1-group"); const destination = text(group?.querySelector("[data-rp1-group-select]")?.value); if (!groupId || !destination) return; const before = engine.get(SCOPE); const matchingKeys = (before.assignments || []) .filter(([, assignment]) => `${assignment.transferMode}|${assignment.destination}` === groupId ) .map(([key]) => key); if (!matchingKeys.length) { console.warn("Reassignment group change found no matching assignments.", { groupId, destination, }); return; } engine.moveGroup(SCOPE, groupId, destination); const after = engine.get(SCOPE); const changed = new Map(after.assignments || []); const synchronized = matchingKeys.every(key => text(changed.get(key)?.destination) === destination ); console.info("Reassignment group destination changed.", { groupId, destination, personnel: matchingKeys.length, synchronized, }); rerender(); }, true); window.$PF.reassignmentV1GroupChangeFix = { version: "1.1.0", active: true, rule: "Group title, dropdown, payload, and preview use AssignmentCore state", }; console.info("Reassignment V1 state-driven group destination fix active."); })();
