// Reassignment V1 — state-driven group destination dropdown // Removes the separate Change button and applies the destination immediately. (function () { "use strict"; const platform = window.PersonnelPlatform; const engine = platform?.engines?.get?.("assignmentCore"); const plugin = window.$PF?.reassignmentV1; if (!platform || !engine || !plugin) return; const SCOPE = "reassignment-v1"; function text(value) { return String(value == null ? "" : value).trim(); } function installStyles() { if (document.getElementById("reassignmentV1GroupDropdownStyles")) return; const style = document.createElement("style"); style.id = "reassignmentV1GroupDropdownStyles"; style.textContent = ` #reassignmentPluginV1Root [data-rp1-group-change] { display: none !important; } #reassignmentPluginV1Root [data-rp1-group-select] { width: min(210px, 100%) !important; cursor: pointer; } `; document.head.appendChild(style); } function rerender() { // open() synchronizes participants and renders entirely from engine state. plugin.close(); plugin.open(); } function updateGroup(select) { const destination = text(select?.value); const groupId = text(select?.dataset.rp1GroupSelect); if (!groupId || !destination) return; const before = engine.get(SCOPE); const matchingKeys = (before.assignments || []) .filter(([, assignment]) => `${assignment.transferMode}|${assignment.destination}` === groupId ) .map(([key]) => key); if (!matchingKeys.length) { console.warn("Reassignment group dropdown found no matching assignments.", { groupId, destination, }); return; } engine.moveGroup(SCOPE, groupId, destination); const after = engine.get(SCOPE); const changed = new Map(after.assignments || []); const synchronized = matchingKeys.every(key => text(changed.get(key)?.destination) === destination ); console.info("Reassignment group destination changed.", { groupId, destination, personnel: matchingKeys.length, synchronized, }); rerender(); } document.addEventListener("change", event => { const select = event.target.closest?.("[data-rp1-group-select]"); if (!select) return; event.preventDefault(); event.stopPropagation(); event.stopImmediatePropagation(); updateGroup(select); }, true); installStyles(); window.$PF.reassignmentV1GroupChangeFix = { version: "1.2.0", active: true, behavior: "dropdown-onchange", }; console.info("Reassignment V1 group destinations now update on dropdown change."); })();
