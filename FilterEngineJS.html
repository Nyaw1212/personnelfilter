function filterPersonnelRecords() { const searchText = String(PersonnelState.filters.search || "").trim(); return PersonnelState.records.filter(record => { if (!recordMatchesGeneralFilters_(record)) { return false; } if (!recordMatchesRanks_(record)) { return false; } if (!recordMatchesSearch_(record, searchText)) { return false; } return true; }); } function recordMatchesGeneralFilters_(record) { return PERSONNEL_APP_CONFIG.filterGroups.every(group => { const selectedValues = PersonnelState.filters[group.key]; if (!selectedValues.size) { return true; } const recordValue = normalizeValue_(record[group.recordKey]); return selectedValues.has(recordValue); }); } function recordMatchesRanks_(record) { const selectedRanks = PersonnelState.filters.ranks; if (!selectedRanks.size) { return true; } return selectedRanks.has( normalizeValue_(record.Rank) ); } function parsePersonnelSearchTerms_(searchText) { return [...new Set( String(searchText || "") .split(/[\n,;]+/) .map(term => term.trim().toLowerCase()) .filter(Boolean) )]; } function recordMatchesSearch_(record, searchText) { const terms = parsePersonnelSearchTerms_(searchText); if (!terms.length) { return true; } const searchableText = PERSONNEL_APP_CONFIG.searchableColumns .map(column => normalizeValue_(record[column]).toLowerCase()) .join(" | "); // Multi-search uses OR logic: a record is included when any pasted term // matches any searchable personnel field. return terms.some(term => searchableText.includes(term)); } function toggleSetValue_(targetSet, value) { const normalizedValue = normalizeValue_(value); if (targetSet.has(normalizedValue)) { targetSet.delete(normalizedValue); return false; } targetSet.add(normalizedValue); return true; } function normalizeValue_(value) { return String(value ?? "").trim(); } function normalizePersonnelName_(value) { return String(value || "") .trim() .replace(/\s+/g, " ") .toUpperCase(); } function analyzePersonnelDuplicates(records) { const nameGroups = new Map(); records.forEach(record => { const displayName = String( record["Full Name"] || "" ).trim(); const normalizedName = normalizePersonnelName_(displayName); if (!normalizedName) { return; } if (!nameGroups.has(normalizedName)) { nameGroups.set(normalizedName, { normalizedName, displayName, records: [], }); } nameGroups .get(normalizedName) .records .push(record); }); const duplicateGroups = [...nameGroups.values()] .filter(group => group.records.length > 1) .sort((a, b) => { if (b.records.length !== a.records.length) { return b.records.length - a.records.length; } return a.displayName.localeCompare( b.displayName, undefined, { sensitivity: "base" } ); }); const duplicateExtraRows = duplicateGroups.reduce( (total, group) => total + group.records.length - 1, 0 ); const duplicateTotalRows = duplicateGroups.reduce( (total, group) => total + group.records.length, 0 ); return { uniqueCount: nameGroups.size, duplicateNameCount: duplicateGroups.length, duplicateExtraRows, duplicateTotalRows, duplicateGroups, };
