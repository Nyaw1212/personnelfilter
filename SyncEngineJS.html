Click Apply Now to sync." ); return; } setSyncStatus( "Changes pending...", "pending" ); setLastSyncLabel( `Waiting ${PERSONNEL_APP_CONFIG.syncDelayMs / 1000} seconds before sync.` ); PersonnelState.syncTimer = setTimeout(() => { applyCurrentResultToSheet(); }, PERSONNEL_APP_CONFIG.syncDelayMs); } async function applyCurrentResultToSheet() { clearTimeout(PersonnelState.syncTimer); const requestNumber = ++PersonnelState.syncRequestNumber; const matchingRows = PersonnelState.filteredRecords .map(record => Number(record.__sheetRow)) .filter(Number.isInteger); setSyncStatus( "Applying to Google Sheets...", "working" ); setLastSyncLabel( "Sending matching sheet rows." ); try { const result = await PersonnelAPI.applyVisibleRows({ matchingRows, }); if ( requestNumber !== PersonnelState.syncRequestNumber ) { return; } setSyncStatus( `Synced · ${result.visibleRows} visible`, "ready" ); setLastSyncLabel( `Mock sync completed at ${formatCurrentTime_()}.` ); } catch (error) { if ( requestNumber !== PersonnelState.syncRequestNumber ) { return; } console.error(error); setSyncStatus( "Sheet sync failed", "error" ); setLastSyncLabel( error?.message || "The sync request could not be completed." ); } } function formatCurrentTime_() { return new Intl.DateTimeFormat( undefined, { hour: "numeric", minute: "2-digit", second: "2-digit", } ).format(new Date()); }
